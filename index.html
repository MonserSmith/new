02 张量索引操作 基础索引

下标从左到右从 0 开始，从右到左从 -1 开始。

data[行下标, 列下标]  ·  data[0轴, 1轴, 2轴] 
import torch
                            torch.manual_seed(0)
                            data = torch.randint(0, 10, (4, 5))
                            print('data->', data)

                            # 行数据：第一行
                            print('data[0] ->', data[0])

                            # 列数据：第一列
                            print('data[:, 0] ->', data[:, 0])

                            # 三维张量索引
                            data2 = torch.randint(0, 10, (3, 4, 5))
                            print('0轴第一个值 ->', data2[0, :, :])
                            print('1轴第一个值 ->', data2[:, 0, :])
                            print('2轴第一个值 ->', data2[:, :, 0])
03 张量索引操作 高级索引 & 切片
列表索引 布尔索引 切片 
# 列表索引：取 (第二行,第三列) 和 (第四行,第五列)
                        print('data[[1,3], [2,4]] ->', data[[1, 3], [2, 4]])

                        # 列表索引 (行列表 & 列列表)
                        print('data[[[1],[3]], [2,4]] ->', data[[[1], [3]], [2, 4]])

                        # 布尔索引：第二列大于6的所有行
                        print(data[:, 1] > 6)
                        print('data[data[:,1] > 6] ->', data[data[:, 1] > 6])

                        # 布尔索引：第三行大于6的所有列
                        print('data[:, data[2] > 6] ->', data[:, data[2] > 6])

                        # 切片：第一行到第三行 (步长2) · 第二列到第四列 (步长2)
                        print('data[::2, 1::2] ->', data[::2, 1::2])
04 形状操作 reshape

reshape(shape) 修改连续或非连续张量的形状，不改变数据。

-1 表示自动计算维度大小 
t1 = torch.randint(0, 10, (5, 6))
                            print('t1形状 ->', t1.shape)   # (5, 6)

                            # 修改为 (2, 15)
                            t2 = t1.reshape((2, 15))
                            print('t2形状 ->', t2.shape)   # (2, 15)

                            # 使用 -1 自动计算
                            t3 = t1.reshape((2, -1))
                            print('t3形状 ->', t3.shape)   # (2, 15)

                            # 三维示例
                            t4 = torch.randint(0, 10, (2, 3, 4))
                            t5 = t4.reshape((-1, 4))
                            print('t5形状 ->', t5.shape)   # (6, 4)
05 形状操作 squeeze / unsqueeze
squeeze 删除值为1的维度
unsqueeze 增加值为1的维度
t1 = torch.randint(0, 10, (3, 1, 2, 1))
                            print('t1形状 ->', t1.shape)   # (3, 1, 2, 1)

                            # squeeze：删除所有值为1的维度
                            t2 = torch.squeeze(t1)
                            print('t2形状 ->', t2.shape)   # (3, 2)

                            # squeeze 指定维度 (维度值不为1不生效)
                            t3 = torch.squeeze(t1, dim=1)
                            print('t3形状 ->', t3.shape)   # (3, 2, 1)

                            # unsqueeze：在指定维度增加维度
                            t4 = t2.unsqueeze(dim=0)
                            print('t4形状 ->', t4.shape)   # (1, 3, 2)

                            t5 = t2.unsqueeze(dim=-1)
                            print('t5形状 ->', t5.shape)   # (3, 2, 1)
06 形状操作 transpose / permute
transpose 交换两个维度
permute 重排任意维度
t1 = torch.randint(0, 10, (3, 4, 5))
                            print('t1形状 ->', t1.shape)   # (3, 4, 5)

                            # transpose：交换 0 和 1 维度
                            t2 = t1.transpose(dim0=1, dim1=0)
                            print('t2形状 ->', t2.shape)   # (4, 3, 5)

                            # permute：重排为 (1, 0, 2)
                            t3 = t1.permute(dims=(1, 0, 2))
                            print('t3形状 ->', t3.shape)   # (4, 3, 5)

                            # permute：重排为 (2, 0, 1)
                            t4 = t1.permute(dims=(2, 0, 1))
                            print('t4形状 ->', t4.shape)   # (5, 3, 4)
07 形状操作 view & contiguous

view() 修改连续张量的形状，reshape() 是更安全的选择。

is_contiguous() 判断是否连续  ·  contiguous() 转为连续 
t1 = torch.randint(0, 10, (3, 4))
                            print('t1连续?', t1.is_contiguous())   # True

                            # view 可用
                            t2 = t1.view((4, 3))
                            print('t2形状 ->', t2.shape)   # (4, 3)

                            # transpose 后变为不连续
                            t3 = t1.transpose(1, 0)
                            print('t3连续?', t3.is_contiguous())   # False

                            # view 在非连续张量上会报错
                            # t3.view((3, 4))  # ❌

                            # 方案1：先 contiguous() 再 view
                            t4 = t3.contiguous().view((3, 4))

                            # 方案2：直接使用 reshape (推荐)
                            t5 = t3.reshape((3, 4))
                            print('t5形状 ->', t5.shape)   # (3, 4)
08 拼接操作 cat / concat

在指定维度上拼接，其他维度必须相同，维度值相加，不改变维度数量。

t1 = torch.randint(0, 10, (2, 3))
                            t2 = torch.randint(0, 10, (2, 3))

                            # dim=0：行方向拼接 (2+2=4 行)
                            t3 = torch.cat([t1, t2], dim=0)
                            print('t3形状 ->', t3.shape)   # (4, 3)

                            # dim=1：列方向拼接 (3+3=6 列)
                            t4 = torch.concat([t1, t2], dim=1)
                            print('t4形状 ->', t4.shape)   # (2, 6)

                            # 三维示例
                            t5 = torch.randint(0, 10, (2, 2, 3))
                            t6 = torch.randint(0, 10, (2, 2, 3))
                            t7 = torch.cat([t5, t6], dim=2)
                            print('t7形状 ->', t7.shape)   # (2, 2, 6)
09 拼接操作 stack

在指定维度上 堆叠，新增一个维度，维度值 = 张量个数。

stack 会 增加维度，cat 不会。 
t1 = torch.randint(0, 10, (2, 3))
                            t2 = torch.randint(0, 10, (2, 3))

                            # dim=0：在第0维堆叠 → (2, 2, 3)
                            t3 = torch.stack([t1, t2], dim=0)
                            print('t3形状 ->', t3.shape)   # (2, 2, 3)

                            # dim=1：在第1维堆叠 → (2, 2, 3)
                            t4 = torch.stack([t1, t2], dim=1)
                            print('t4形状 ->', t4.shape)   # (2, 2, 3)

                            # dim=-1：在最后维度堆叠 → (2, 3, 2)
                            t5 = torch.stack([t1, t2], dim=-1)
                            print('t5形状 ->', t5.shape)   # (2, 3, 2)

                            # 堆叠多个张量 (3个)
                            t6 = torch.stack([t1, t2, t1], dim=0)
                            print('t6形状 ->', t6.shape)   # (3, 2, 3)
10 自动微分 梯度计算

requires_grad=True 开启自动微分，.backward() 计算梯度。

⚠️ 梯度默认 累加，每次反向传播前需 zero_() 清零。 
w = torch.tensor([10, 20], requires_grad=True, dtype=torch.float32)
                            loss = 2 * w ** 2

                            # loss 必须是标量，用 sum() 聚合
                            loss.sum().backward()

                            # 查看梯度
                            print('w.grad ->', w.grad)   # [40, 80]

                            # 梯度更新 (梯度下降)
                            lr = 0.01
                            w.data = w.data - lr * w.grad

                            # 梯度清零 (避免累加)
                            if w.grad is not None:
                            w.grad.zero_()
11 自动微分 梯度下降法求最优解

通过迭代更新权重，使损失函数 loss = w² + 20 最小化。

w = torch.tensor(10, requires_grad=True, dtype=torch.float32)
                            lr = 0.01

                            for i in range(1, 1001):
                            # 前向传播：计算损失
                            loss = w ** 2 + 20

                            # 梯度清零
                            if w.grad is not None:
                            w.grad.zero_()

                            # 反向传播：计算梯度
                            loss.sum().backward()

                            # 梯度更新：W = W - lr * grad
                            w.data = w.data - lr * w.grad

                            if i % 200 == 0:
                            print(f'第{i:4d}次: w={w.item():.6f}, loss={loss.item():.6f}')

                            # 最终 w → 0, loss → 20
12 自动微分 注意点 · detach

自动微分的张量 不能 直接转换为 NumPy 数组，需用 .detach() 剥离。

x1 = torch.tensor(10, requires_grad=True, dtype=torch.float32)
                            print('x1.requires_grad ->', x1.requires_grad)   # True

                            # detach()：生成新的张量，不自动微分，数据共享
                            x2 = x1.detach()
                            print('x2.requires_grad ->', x2.requires_grad)   # False

                            # 数据地址相同 (共享存储)
                            print(id(x1.data) == id(x2.data))   # True

                            # 转换为 NumPy 数组 (需先 detach)
                            n1 = x2.numpy()
                            print('n1 ->', n1)   # 10.0

                            # ❌ 直接转换会报错
                            # n2 = x1.numpy()  # RuntimeError
13 自动微分 应用 · MSE 损失

使用 torch.nn.MSELoss 计算均方误差损失，并自动计算梯度。

import torch.nn as nn

                            # 1. 定义样本 x 和 y
                            x = torch.ones((2, 5))
                            y = torch.zeros((2, 3))

                            # 2. 模型权重 (自动微分)
                            w = torch.randn((5, 3), requires_grad=True)
                            b = torch.randn((3,), requires_grad=True)

                            # 3. 前向传播：计算预测值
                            y_pred = torch.matmul(x, w) + b

                            # 4. MSE 损失
                            criterion = nn.MSELoss()
                            loss = criterion(y_pred, y)

                            # 5. 反向传播：计算梯度
                            loss.sum().backward()

                            print('w.grad 形状 ->', w.grad.shape)   # (5, 3)
                            print('b.grad ->', b.grad)             # (3,)
